<script>
(function() {
const canvas = document.getElementById('genCanvas');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  let W, H, particles = [], t = 0;

  function resize() {
    const dpr = window.devicePixelRatio || 1;
    W = canvas.offsetWidth;
    H = canvas.offsetHeight;
    canvas.width = W * dpr;
    canvas.height = H * dpr;
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.scale(dpr, dpr);
  }
  resize();
  window.addEventListener('resize', resize);

  // Data-flow particles
  class Particle {
    constructor() { this.reset(); }
    reset() {
      this.x = Math.random() * W;
      this.y = Math.random() * H;
      this.vx = (Math.random() - 0.5) * 0.15;
      this.vy = (Math.random() - 0.5) * 0.15;
      this.life = Math.random();
      this.maxLife = 0.004 + Math.random() * 0.003;
      this.size = 1 + Math.random() * 2.5;
      this.trail = [];
    }
    update() {
      // Noise-like steering
      const angle = Math.sin(this.x * 0.008 + t * 0.3) * Math.cos(this.y * 0.006 + t * 0.2) * Math.PI * 2;
      this.vx += Math.cos(angle) * 0.06;
      this.vy += Math.sin(angle) * 0.06;
      const speed = Math.sqrt(this.vx * this.vx + this.vy * this.vy);
      if (speed > 1.8) { this.vx *= 1.8 / speed; this.vy *= 1.8 / speed; }
      this.x += this.vx;
      this.y += this.vy;
      this.trail.push({ x: this.x, y: this.y });
      if (this.trail.length > 18) this.trail.shift();
      this.life += this.maxLife;
      if (this.life > 1 || this.x < 0 || this.x > W || this.y < 0 || this.y > H) this.reset();
    }
    draw() {
      if (this.trail.length < 2) return;
      const alpha = Math.sin(this.life * Math.PI) * 0.65;
      ctx.beginPath();
      ctx.moveTo(this.trail[0].x, this.trail[0].y);
      for (let i = 1; i < this.trail.length; i++) ctx.lineTo(this.trail[i].x, this.trail[i].y);
      ctx.strokeStyle = `rgba(232,248,0,${alpha * 0.5})`;
      ctx.lineWidth = 1;
      ctx.stroke();
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size * 0.5, 0, Math.PI * 2);
      ctx.fillStyle = `rgba(232,248,0,${alpha})`;
      ctx.fill();
    }
  }

  for (let i = 0; i < 110; i++) particles.push(new Particle());

  // Concentric rings in background
  function drawRings() {
    const cx = W * 0.55, cy = H * 0.5;
    for (let r = 40; r < Math.max(W, H); r += 70) {
      ctx.beginPath();
      ctx.arc(cx, cy, r + Math.sin(t * 0.5 + r * 0.01) * 8, 0, Math.PI * 2);
      ctx.strokeStyle = `rgba(232,248,0,${0.025 - r * 0.00004})`;
      ctx.lineWidth = 0.5;
      ctx.stroke();
    }
  }

  function frame() {
    ctx.fillStyle = 'rgba(10,10,10,0.18)';
    ctx.fillRect(0, 0, W, H);
    drawRings();
    particles.forEach(p => { p.update(); p.draw(); });
    t += 0.012;
    requestAnimationFrame(frame);
  }
  frame();
})();
</script>

(function() {
const canvas = document.getElementById('campaignCanvas');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  let W, H, nodes = [], t = 0;

  function resize() {
    const dpr = window.devicePixelRatio || 1;
    W = canvas.offsetWidth;
    H = canvas.offsetHeight;
    canvas.width = W * dpr;
    canvas.height = H * dpr;
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.scale(dpr, dpr);
  }
  resize();
  window.addEventListener('resize', resize);
// Fixed data nodes
  for (let i = 0; i < 22; i++) {
    nodes.push({
      x: 60 + Math.random() * (W - 120),
      y: 60 + Math.random() * (H - 120),
      r: 2 + Math.random() * 5,
      pulse: Math.random() * Math.PI * 2
    });
  }

  // Moving data packets along edges
  let packets = [];
  function spawnPacket() {
    const a = Math.floor(Math.random() * nodes.length);
    let b = Math.floor(Math.random() * nodes.length);
    while (b === a) b = Math.floor(Math.random() * nodes.length);
    const dist = Math.hypot(nodes[a].x - nodes[b].x, nodes[a].y - nodes[b].y);
    if (dist > 280) return;
    packets.push({ a, b, t: 0, speed: 0.006 + Math.random() * 0.006 });
  }
  for (let i = 0; i < 12; i++) spawnPacket();

  function frame() {
    ctx.fillStyle = 'rgba(10,10,10,0.22)';
    ctx.fillRect(0, 0, W, H);

    // Draw edges between close nodes
    for (let i = 0; i < nodes.length; i++) {
      for (let j = i + 1; j < nodes.length; j++) {
        const d = Math.hypot(nodes[i].x - nodes[j].x, nodes[i].y - nodes[j].y);
        if (d < 200) {
          ctx.beginPath();
          ctx.moveTo(nodes[i].x, nodes[i].y);
          ctx.lineTo(nodes[j].x, nodes[j].y);
          ctx.strokeStyle = `rgba(232,248,0,${0.08 * (1 - d / 200)})`;
          ctx.lineWidth = 0.5;
          ctx.stroke();
        }
      }
    }

    // Draw nodes
    nodes.forEach((n, i) => {
      n.pulse += 0.04;
      const alpha = 0.5 + Math.sin(n.pulse) * 0.3;
      ctx.beginPath();
      ctx.arc(n.x, n.y, n.r, 0, Math.PI * 2);
      ctx.fillStyle = `rgba(232,248,0,${alpha})`;
      ctx.fill();
      // Pulse ring
      const pr = n.r + 6 + Math.sin(n.pulse) * 4;
      ctx.beginPath();
      ctx.arc(n.x, n.y, pr, 0, Math.PI * 2);
      ctx.strokeStyle = `rgba(232,248,0,${0.15 * alpha})`;
      ctx.lineWidth = 0.8;
      ctx.stroke();
    });

    // Draw packets
    packets.forEach((p, idx) => {
      p.t += p.speed;
      if (p.t >= 1) { packets.splice(idx, 1); spawnPacket(); return; }
      const na = nodes[p.a], nb = nodes[p.b];
      const x = na.x + (nb.x - na.x) * p.t;
      const y = na.y + (nb.y - na.y) * p.t;
      ctx.beginPath();
      ctx.arc(x, y, 3, 0, Math.PI * 2);
      ctx.fillStyle = '#e8f800';
      ctx.fill();
    });

    t += 0.01;
    requestAnimationFrame(frame);
  }
  frame();
})();
</script>

(function() {
const container = document.getElementById('sonicVis');
  if (!container) return;
  const rowCount = 8;
  const barsPerRow = 64;
  const rows = [];
  for (let r = 0; r < rowCount; r++) {
    const row = document.createElement('div');
    row.className = 'waveform-row';
    const bars = [];
    for (let b = 0; b < barsPerRow; b++) {
      const bar = document.createElement('div');
      bar.className = 'wave-bar';
      const baseH = 6 + Math.sin((b / barsPerRow) * Math.PI) * 20;
      bar.style.height = baseH + 'px';
      bar.style.animationDelay = (b * 30 + r * 80) + 'ms';
      bar.style.animationDuration = (1.4 + Math.random() * 0.8) + 's';
      const alpha = 0.3 + (rowCount - r) / rowCount * 0.6;
      bar.style.opacity = alpha;
      row.appendChild(bar);
      bars.push(bar);
    }
    container.appendChild(row);
    rows.push(bars);
  }

  // Animate heights based on sine pattern over time
  let t = 0;
  function animate() {
    rows.forEach((bars, r) => {
      bars.forEach((bar, b) => {
        const h = 4 + Math.abs(Math.sin((b / barsPerRow) * Math.PI * 3 + t + r * 0.3)) * 24;
        bar.style.height = h + 'px';
      });
    });
    t += 0.05;
    requestAnimationFrame(animate);
  }
  animate();
})();

(function() {
const canvas = document.getElementById('installCanvas');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  let W, H, t = 0;

  function resize() {
    const dpr = window.devicePixelRatio || 1;
    W = canvas.offsetWidth;
    H = canvas.offsetHeight;
    canvas.width = W * dpr;
    canvas.height = H * dpr;
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.scale(dpr, dpr);
  }
  resize();
  window.addEventListener('resize', resize);
// Grid of cells that respond to a wave
  const cols = 18, rows = 12;

  function frame() {
    ctx.fillStyle = 'rgba(10,10,10,0.25)';
    ctx.fillRect(0, 0, W, H);

    const cw = W / cols, ch = H / rows;

    for (let c = 0; c < cols; c++) {
      for (let r = 0; r < rows; r++) {
        const cx = (c + 0.5) * cw, cy = (r + 0.5) * ch;
        const dist = Math.hypot(c - cols / 2, r - rows / 2);
        const wave = Math.sin(dist * 0.6 - t * 1.8) * 0.5 + 0.5;
        const wave2 = Math.sin(c * 0.4 + t * 0.7) * Math.cos(r * 0.5 - t * 0.5) * 0.5 + 0.5;
        const v = wave * 0.6 + wave2 * 0.4;
        const size = 1 + v * (Math.min(cw, ch) * 0.38);
        const alpha = 0.1 + v * 0.7;

        ctx.beginPath();
        // Alternate between square and circle
        if ((c + r) % 3 === 0) {
          ctx.arc(cx, cy, size * 0.5, 0, Math.PI * 2);
        } else {
          ctx.rect(cx - size * 0.4, cy - size * 0.4, size * 0.8, size * 0.8);
        }
        ctx.fillStyle = `rgba(232,248,0,${alpha})`;
        ctx.fill();
      }
    }

    t += 0.018;
    requestAnimationFrame(frame);
  }
  frame();
})();

(function() {
const container = document.getElementById('motionVis');
  if (!container) return;
  
  container.innerHTML = '<canvas id="motionCanvas"></canvas>';
  const canvas = document.getElementById('motionCanvas');
  const ctx = canvas.getContext('2d');
  
  let w, h;
  const resize = () => {
    w = container.offsetWidth;
    h = container.offsetHeight;
    canvas.width = w * window.devicePixelRatio;
    canvas.height = h * window.devicePixelRatio;
    ctx.scale(window.devicePixelRatio, window.devicePixelRatio);
  };
  window.addEventListener('resize', resize);
  resize();
  window.addEventListener('resize', resize);
const particles = [];
  const particleCount = 45;
  for(let i=0; i < particleCount; i++) {
    particles.push({
      x: Math.random() * w,
      y: Math.random() * h,
      vx: (Math.random() - 0.5) * 0.4,
      vy: (Math.random() - 0.5) * 0.4,
      r: Math.random() * 1.5 + 0.5,
      alpha: Math.random() * 0.5 + 0.2
    });
  }

  let time = 0;
  function animate() {
    time += 0.008;
    ctx.fillStyle = '#080808';
    ctx.fillRect(0, 0, w, h);

    // subtle grid
    ctx.strokeStyle = 'rgba(232,248,0,0.03)';
    ctx.lineWidth = 0.5;
    const gSize = 50;
    for(let x=0; x<w; x+=gSize) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, h); ctx.stroke(); }
    for(let y=0; y<h; y+=gSize) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(w, y); ctx.stroke(); }

    // Renaissance Sacred Geometry (Harmonic Oscillations)
    ctx.save();
    ctx.translate(w/2, h/2);
    ctx.strokeStyle = 'rgba(232,248,0,0.08)';
    ctx.lineWidth = 1;
    for(let i=0; i<6; i++) {
        ctx.save();
        ctx.rotate(time * 0.2 + (i * Math.PI / 3));
        const orbitSize = 60 + i * 20;
        ctx.beginPath();
        ctx.ellipse(orbitSize * Math.cos(time * 0.5), 0, orbitSize, orbitSize * 0.6, time, 0, Math.PI * 2);
        ctx.stroke();
        ctx.restore();
    }
    ctx.restore();

    // Data Mesh (Cyber)
    particles.forEach((p, i) => {
      p.x += p.vx; p.y += p.vy;
      if(p.x < 0 || p.x > w) p.vx *= -1;
      if(p.y < 0 || p.y > h) p.vy *= -1;

      ctx.fillStyle = `rgba(232,248,0, ${p.alpha})`;
      ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2); ctx.fill();

      // Connections
      for(let j=i+1; j<particles.length; j++) {
        const p2 = particles[j];
        const d = Math.hypot(p.x - p2.x, p.y - p2.y);
        if(d < 90) {
          ctx.strokeStyle = `rgba(232,248,0, ${0.15 * (1 - d/90)})`;
          ctx.lineWidth = 0.5;
          ctx.beginPath(); ctx.moveTo(p.x, p.y); ctx.lineTo(p2.x, p2.y); ctx.stroke();
        }
      }
    });

    // Floating Hex Data (Cyber)
    if(Math.random() > 0.97) {
        ctx.font = '8px DM Mono';
        ctx.fillStyle = 'rgba(232,248,0,0.2)';
        const hex = '0x' + Math.floor(Math.random()*0xffffff).toString(16).toUpperCase();
        ctx.fillText(hex, Math.random() * w, Math.random() * h);
    }

    requestAnimationFrame(animate);
  }
  animate();
})();
